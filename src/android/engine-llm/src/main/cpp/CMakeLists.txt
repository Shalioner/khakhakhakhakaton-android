cmake_minimum_required(VERSION 3.22.1)
project(neurollm CXX C)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# llama.cpp как статическая библиотека: без сети, без OpenMP/llamafile, без tools/examples/tests.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_HTML OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)

# armv8.2 dotprod/fp16 — есть на всех arm64-телефонах последних ~6 лет; ускоряет Q4_K матмул.
if (ANDROID_ABI STREQUAL "arm64-v8a")
    set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "" FORCE)
endif()

add_subdirectory(llama.cpp)

add_library(neurollm SHARED llm_jni.cpp)
if (ANDROID)
    target_link_libraries(neurollm PRIVATE llama ggml android log)
    # Выравнивание сегментов 16 КБ: устройства Android 15+ со страницами 16 КБ иначе не загрузят .so.
    target_link_options(neurollm PRIVATE "-Wl,-z,max-page-size=16384")
else()
    find_package(JNI REQUIRED)
    target_include_directories(neurollm PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(neurollm PRIVATE llama ggml)
endif()
if (MSVC)
    target_compile_options(neurollm PRIVATE /utf-8 /O2)
else()
    target_compile_options(neurollm PRIVATE -O3)
endif()
